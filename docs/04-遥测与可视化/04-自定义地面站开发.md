```cpp
class MavlinkManager : public QObject {
    Q_OBJECT

    Q_PROPERTY(float roll READ getRoll NOTIFY attitudeChanged)
    Q_PROPERTY(float pitch READ getPitch NOTIFY attitudeChanged)
    Q_PROPERTY(float yaw READ getYaw NOTIFY attitudeChanged)
    Q_PROPERTY(double latitude READ getLatitude NOTIFY positionChanged)
    Q_PROPERTY(double longitude READ getLongitude NOTIFY positionChanged)
    Q_PROPERTY(float altitude READ getAltitude NOTIFY positionChanged)

public:
    explicit MavlinkManager(QObject *parent = nullptr);
    
    bool isConnected() const { return m_connected; }
    float getRoll() const { return m_roll; }
    float getPitch() const { return m_pitch; }
    float getYaw() const { return m_yaw; }
    double getLatitude() const { return m_latitude; }
    double getLongitude() const { return m_longitude; }
    float getAltitude() const { return m_altitude; }
    
    Q_INVOKABLE void connectToVehicle(const QString &host, int port);
    Q_INVOKABLE void arm();
    Q_INVOKABLE void disarm();
    Q_INVOKABLE void takeoff(float altitude);
    Q_INVOKABLE void land();
    Q_INVOKABLE void returnToLaunch();

signals:
    void connectionChanged();
    void attitudeChanged();
    void positionChanged();
    void batteryChanged(float voltage, float percent);
    void messageReceived(const QString &type, const QString &content);

private slots:
    void readPendingDatagrams();
    void checkConnection();

private:
    void processMessage(const mavlink_message_t &msg);
    
    QUdpSocket *m_socket;
    QTimer *m_heartbeatTimer;
    bool m_connected = false;
    
    float m_roll = 0, m_pitch = 0, m_yaw = 0;

    float m_altitude = 0;
```

```cpp
// mavlink_manager.cpp
#include "mavlink_manager.h"

MavlinkManager::MavlinkManager(QObject *parent) : QObject(parent) {
    m_socket = new QUdpSocket(this);
    connect(m_socket, &QUdpSocket::readyRead, this, &MavlinkManager::readPendingDatagrams);
    
    m_heartbeatTimer = new QTimer(this);
    connect(m_heartbeatTimer, &QTimer::timeout, this, &MavlinkManager::checkConnection);
    m_heartbeatTimer->start(1000);
}

void MavlinkManager::connectToVehicle(const QString &host, int port) {
    m_socket->bind(QHostAddress::Any, port);
    m_connected = true;
    emit connectionChanged();
}

void MavlinkManager::readPendingDatagrams() {
    while (m_socket->hasPendingDatagrams()) {
        QByteArray data = m_socket->readDatagram(m_socket->pendingDatagramSize());
        
        mavlink_message_t msg;
        mavlink_status_t status;
        
        for (int i = 0; i < data.size(); i++) {
            if (mavlink_parse_char(MAVLINK_COMM_0, data[i], &msg, &status)) {
                processMessage(msg);
            }
        }
    }
}

void MavlinkManager::processMessage(const mavlink_message_t &msg) {
    switch (msg.msgid) {
        case MAVLINK_MSG_ID_HEARTBEAT:
            // 处理心跳
            break;
            
        case MAVLINK_MSG_ID_ATTITUDE: {
            mavlink_attitude_t att;
            mavlink_msg_attitude_decode(&msg, &att);
            m_roll = att.roll * 180.0f / M_PI;
            m_pitch = att.pitch * 180.0f / M_PI;
            m_yaw = att.yaw * 180.0f / M_PI;
            emit attitudeChanged();
            break;
        }
        
        case MAVLINK_MSG_ID_GLOBAL_POSITION_INT: {
            mavlink_global_position_int_t pos;
            mavlink_msg_global_position_int_decode(&msg, &pos);
            m_latitude = pos.lat / 1e7;
            m_longitude = pos.lon / 1e7;
            m_altitude = pos.relative_alt / 1000.0f;
            emit positionChanged();
            break;
        }
        
        case MAVLINK_MSG_ID_BATTERY_STATUS: {
            mavlink_battery_status_t bat;
            mavlink_msg_battery_status_decode(&msg, &bat);
            m_batteryVoltage = bat.voltages[0] / 1000.0f;
            m_batteryPercent = bat.battery_remaining;
            emit batteryChanged(m_batteryVoltage, m_batteryPercent);
            break;
        }
    }
}

void MavlinkManager::arm() {
    mavlink_message_t msg;
    mavlink_msg_command_long_pack(
        255, 190, &msg,
        1, 1,
        MAV_CMD_COMPONENT_ARM_DISARM,
        0, 1, 0, 0, 0, 0, 0, 0
    );
    QByteArray data((char*)&msg, sizeof(msg));
    m_socket->writeDatagram(data, QHostAddress("192.168.1.100"), 14550);
}

void MavlinkManager::takeoff(float altitude) {
    mavlink_message_t msg;
    mavlink_msg_command_long_pack(
        255, 190, &msg,
        1, 1,
        MAV_CMD_NAV_TAKEOFF,
        0, 0, 0, 0, 0, 0, 0, altitude
    );
    QByteArray data((char*)&msg, sizeof(msg));
    m_socket->writeDatagram(data, QHostAddress("192.168.1.100"), 14550);
}
```

### 3.2 WebSocket 服务器

```python
# Python WebSocket 服务器
import asyncio
import websockets
from pymavlink import mavutil
import json

class WebGCS:
    def __init__(self):
        self.clients = set()
        self.mavlink_conn = None
    
    async def connect_mavlink(self, connection_string):
        """连接 MAVLink"""
        self.mavlink_conn = mavutil.mavlink_connection(connection_string)
        self.mavlink_conn.wait_heartbeat()
    
    async def handle_client(self, websocket, path):
        """处理 WebSocket 客户端"""
        self.clients.add(websocket)
        try:
            async for message in websocket:
                data = json.loads(message)
                await self.process_command(data)
        finally:
            self.clients.remove(websocket)
    
    async def process_command(self, command):
        """处理客户端命令"""
        if command['type'] == 'arm':
            self.mavlink_conn.mav.command_long_send(
                1, 1, 400, 0, 1, 0, 0, 0, 0, 0, 0
            )
        elif command['type'] == 'takeoff':
            alt = command.get('altitude', 10)
            self.mavlink_conn.mav.command_long_send(
                1, 1, 22, 0, 0, 0, 0, 0, 0, 0, alt
            )
    
    async def broadcast_telemetry(self):
        """广播遥测数据"""
        while True:
            msg = self.mavlink_conn.recv_match(blocking=False)
            if msg:
                data = self.parse_message(msg)
                if data:
                    for client in self.clients:
                        await client.send(json.dumps(data))
            await asyncio.sleep(0.01)
    
    def parse_message(self, msg):
        """解析 MAVLink 消息"""
        msg_type = msg.get_type()
        
        if msg_type == 'ATTITUDE':
            return {
                'type': 'attitude',
                'roll': msg.roll,
                'pitch': msg.pitch,
                'yaw': msg.yaw
            }
        elif msg_type == 'GLOBAL_POSITION_INT':
            return {
                'type': 'position',
                'lat': msg.lat / 1e7,
                'lon': msg.lon / 1e7,
                'alt': msg.relative_alt / 1000
            }
        return None

# 启动服务器
async def main():
    gcs = WebGCS()
    await gcs.connect_mavlink('udp:127.0.0.1:14550')
    
    server = await websockets.serve(gcs.handle_client, '0.0.0.0', 8765)
    await gcs.broadcast_telemetry()

asyncio.run(main())
```

```javascript
// React 组件示例
import React, { useState, useEffect } from 'react';
import { MapContainer, TileLayer, Marker, Popup } from 'react-leaflet';

function GroundStation() {
    const [position, setPosition] = useState({ lat: 30.123, lon: 120.456 });
    const [attitude, setAttitude] = useState({ roll: 0, pitch: 0, yaw: 0 });
    const [ws, setWs] = useState(null);
    
    useEffect(() => {
        const socket = new WebSocket('ws://localhost:8765');
        
        socket.onmessage = (event) => {
            const data = JSON.parse(event.data);
            
            if (data.type === 'position') {
                setPosition({ lat: data.lat, lon: data.lon });
            } else if (data.type === 'attitude') {
                setAttitude({
                    roll: data.roll * 180 / Math.PI,
                    pitch: data.pitch * 180 / Math.PI,
                    yaw: data.yaw * 180 / Math.PI
                });
            }
        };
        
        setWs(socket);
        return () => socket.close();
    }, []);
    
    const sendCommand = (command) => {
        if (ws) {
            ws.send(JSON.stringify(command));
        }
    };
    
    return (
        <div className="ground-station">
            <div className="sidebar">
                <h2>姿态</h2>
                <p>Roll: {attitude.roll.toFixed(1)}°</p>
                <p>Pitch: {attitude.pitch.toFixed(1)}°</p>
                <p>Yaw: {attitude.yaw.toFixed(1)}°</p>
                
                <h2>控制</h2>
                <button onClick={() => sendCommand({type: 'arm'})}>解锁</button>
                <button onClick={() => sendCommand({type: 'takeoff', altitude: 10})}>起飞</button>
                <button onClick={() => sendCommand({type: 'land'})}>降落</button>
                <button onClick={() => sendCommand({type: 'rtl'})}>返航</button>
            </div>
            
            <div className="map">
                <MapContainer center={[position.lat, position.lon]} zoom={15}>
                    <TileLayer url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" />
                    <Marker position={[position.lat, position.lon]}>
                        <Popup>无人机位置</Popup>
                    </Marker>
                </MapContainer>
            </div>
        </div>
    );
}
```

```qml
// AttitudeIndicator.qml
import QtQuick 2.15

Item {
    width: 200
    height: 200
    
    property real roll: 0
    property real pitch: 0
    
    Canvas {
        anchors.fill: parent
        onPaint: {
            var ctx = getContext("2d");
            ctx.reset();
            
            // 绘制背景
            ctx.fillStyle = "#1a1a2e";
            ctx.fillRect(0, 0, width, height);
            
            // 绘制地平线
            ctx.save();
            ctx.translate(width/2, height/2);
            ctx.rotate(-roll * Math.PI / 180);
            
            // 天空
            ctx.fillStyle = "#4a90d9";
            ctx.fillRect(-width, -height, width*2, height + pitch);
            
            // 地面
            ctx.fillStyle = "#8b4513";
            ctx.fillRect(-width, pitch, width*2, height);
            
            ctx.restore();
            
            // 绘制飞机符号
            ctx.strokeStyle = "#00ff00";
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.moveTo(width/2 - 30, height/2);
            ctx.lineTo(width/2 - 10, height/2);
            ctx.lineTo(width/2, height/2 + 10);
            ctx.lineTo(width/2 + 10, height/2);
            ctx.lineTo(width/2 + 30, height/2);
            ctx.stroke();
        }
    }
    
    onRollChanged: canvas.requestPaint()
    onPitchChanged: canvas.requestPaint()
}
```

<details>
<summary>参考答案</summary>

**1. 技术栈对比：**

Qt/QML：
- 优点：高性能，原生体验
- 缺点：学习曲线陡峭，跨平台一致性差

Web：
- 优点：跨平台，易部署，生态丰富
- 缺点：性能较低，离线支持差

**2. 实时推送方案：**

- WebSocket：双向实时通信
- Server-Sent Events：服务器单向推送
- 定时轮询：简单但效率低

**3. 核心功能模块：**

- MAVLink 连接管理
- 遥测数据接收与解析
- 地图显示与航点规划
- 仪表盘与数据可视化
- 命令发送与任务管理
- 日志记录与回放

**4. 多机监控设计：**

- 每架无人机独立连接
- 统一消息队列处理
- 多标签页/多窗口显示
- 无人机选择与切换

**5. 稳定性保障：**

- 异常处理与重连机制
- 心跳检测与超时处理
- 数据校验与错误恢复
- 日志记录与问题追踪

</details>
