# 使用 pymavlink 读取 BIN 日志
from pymavlink import mavutil

def read_bin_log(filename):
    """读取 BIN 日志文件"""
    log = mavutil.mavlink_connection(filename, dialect='ardupilotmega')
    
    messages = []
    while True:
        msg = log.recv_match(blocking=False)
        if msg is None:
            break
        messages.append(msg)
    
    return messages

# 提取特定数据
def extract_attitude(messages):
    """提取姿态数据"""
    attitude_data = []
    for msg in messages:
        if msg.get_type() == 'ATT':
            attitude_data.append({
                'time': msg.TimeUS,
                'roll': msg.Roll,
                'pitch': msg.Pitch,
                'yaw': msg.Yaw
            })
    return attitude_data
```

# 使用 pyulog 读取 ULog 日志
from pyulog import ULog

def read_ulog(filename):
    """读取 ULog 文件"""
    ulog = ULog(filename)
    
    # 获取所有数据集
    datasets = ulog.data_list
    print(f"数据集数量: {len(datasets)}")
    
    # 获取姿态数据
    att_data = ulog.get_dataset('vehicle_attitude')
    roll = att_data.data['roll']
    pitch = att_data.data['pitch']
    yaw = att_data.data['yaw']
    timestamp = att_data.data['timestamp']
    
    return {
        'timestamp': timestamp,
        'roll': roll,
        'pitch': pitch,
        'yaw': yaw
    }

# 提取位置数据
def extract_position(ulog):
    """提取位置数据"""
    pos_data = ulog.get_dataset('vehicle_local_position')
    return {
        'x': pos_data.data['x'],
        'y': pos_data.data['y'],
        'z': pos_data.data['z'],
        'timestamp': pos_data.data['timestamp']
    }
```

## 4. 数据导出与处理

### 4.1 导出为 CSV

```python
import pandas as pd

def export_to_csv(messages, output_file):
    """将日志导出为 CSV"""
    data = []
    
    for msg in messages:
        msg_type = msg.get_type()
        
        if msg_type == 'ATT':
            data.append({
                'time_us': msg.TimeUS,
                'type': 'ATT',
                'roll': msg.Roll,
                'pitch': msg.Pitch,
                'yaw': msg.Yaw
            })
        elif msg_type == 'GPS':
            data.append({
                'time_us': msg.TimeUS,
                'type': 'GPS',
                'lat': msg.Lat,
                'lon': msg.Lng,
                'alt': msg.Alt,
                'sats': msg.NSats
            })
        elif msg_type == 'BAT':
            data.append({
                'time_us': msg.TimeUS,
                'type': 'BAT',
                'voltage': msg.Volt,
                'current': msg.Curr,
                'consumed': msg.CurrTot
            })
    
    df = pd.DataFrame(data)
    df.to_csv(output_file, index=False)
    return df
```

### 4.2 数据清洗

```python
def clean_flight_data(df):
    """清洗飞行数据"""
    # 删除重复数据
    df = df.drop_duplicates()
    
    # 删除无效数据
    df = df.dropna()
    
    # 时间排序
    df = df.sort_values('time_us')
    
    # 重置索引
    df = df.reset_index(drop=True)
    
    # 计算相对时间（秒）
    df['time_s'] = (df['time_us'] - df['time_us'][0]) / 1e6
    
    return df
```

def analyze_pid_performance(df):
    """分析 PID 控制性能"""
    ctun_data = df[df['type'] == 'CTUN']
    
    # 计算跟踪误差
    ctun_data = ctun_data.copy()
    ctun_data['alt_error'] = ctun_data['Alt'] - ctun_data['DAlt']
    
    # 计算统计指标
    error_mean = ctun_data['alt_error'].mean()
    error_std = ctun_data['alt_error'].std()
    error_max = ctun_data['alt_error'].abs().max()
    
    print(f"高度跟踪误差:")
    print(f"  平均误差: {error_mean:.2f}m")
    print(f"  标准差: {error_std:.2f}m")
    print(f"  最大误差: {error_max:.2f}m")
    
    # 绘制误差
    fig, ax = plt.subplots(figsize=(12, 4))
    ax.plot(ctun_data['time_s'], ctun_data['alt_error'], 'b-')
    ax.axhline(y=0, color='r', linestyle='--')
    ax.set_xlabel('时间 (s)')
    ax.set_ylabel('高度误差 (m)')
    ax.set_title('高度跟踪误差')

def analyze_motor_balance(df):
    """分析电机平衡"""
    mot_data = df[df['type'] == 'MOT']
    
    # 提取电机输出
    motors = ['Mot1', 'Mot2', 'Mot3', 'Mot4', 'Mot5', 'Mot6', 'Mot7', 'Mot8']
    available_motors = [m for m in motors if m in mot_data.columns]
    
    if not available_motors:
        print("无电机数据")
        return
    
    # 计算统计
    motor_stats = mot_data[available_motors].describe()
    print("电机输出统计:")
    print(motor_stats)
    
    # 计算电机间差异
    motor_mean = mot_data[available_motors].mean()
    motor_diff = motor_mean.max() - motor_mean.min()
    
    print(f"\n电机间平均差异: {motor_diff:.1f}")
    if motor_diff > 100:
        print("警告: 电机输出差异较大，可能需要校准")

## 7. 完整分析流程

```python
#!/usr/bin/env python3
"""飞行日志完整分析"""

import sys
import pandas as pd
from pymavlink import mavutil

def full_analysis(log_file):
    """完整飞行分析"""
    print(f"分析日志文件: {log_file}")
    
    # 1. 读取日志
    log = mavutil.mavlink_connection(log_file, dialect='ardupilotmega')
    messages = []
    while True:
        msg = log.recv_match(blocking=False)
        if msg is None:
            break
        messages.append(msg)
    
    print(f"共读取 {len(messages)} 条消息")
    
    # 2. 导出为 DataFrame
    df = export_to_csv(messages, 'temp.csv')
    df = clean_flight_data(df)
    
    # 3. 飞行统计
    flight_time = df['time_s'].max() - df['time_s'].min()
    print(f"\n飞行统计:")
    print(f"  飞行时间: {flight_time:.1f}秒 ({flight_time/60:.1f}分钟)")
    
    # 4. 各类分析
    print("\n=== 姿态分析 ===")
    att_data = df[df['type'] == 'ATT']
    if not att_data.empty:
        print(f"  Roll 范围: {att_data['roll'].min():.1f}° 到 {att_data['roll'].max():.1f}°")
        print(f"  Pitch 范围: {att_data['pitch'].min():.1f}° 到 {att_data['pitch'].max():.1f}°")
    
    print("\n=== GPS 分析 ===")
    gps_data = df[df['type'] == 'GPS']
    if not gps_data.empty:
        print(f"  平均卫星数: {gps_data['sats'].mean():.1f}")
        print(f"  最大高度: {gps_data['alt'].max():.1f}m")
    
    print("\n=== 电池分析 ===")
    bat_data = df[df['type'] == 'BAT']
    if not bat_data.empty:
        print(f"  起始电压: {bat_data['voltage'].iloc[0]:.1f}V")
        print(f"  结束电压: {bat_data['voltage'].iloc[-1]:.1f}V")
        print(f"  电压降: {bat_data['voltage'].iloc[0] - bat_data['voltage'].iloc[-1]:.1f}V")
    
    # 5. 绘制图表
    print("\n生成图表...")
    plot_flight_path(df)
    plot_attitude(df)
    plot_battery(df)
    
    print("\n分析完成!")

if __name__ == '__main__':
    if len(sys.argv) > 1:
        full_analysis(sys.argv[1])
    else:
        print("用法: python flight_analysis.py <log_file.bin>")
```
